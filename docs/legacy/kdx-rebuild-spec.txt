# KDX Modern Rebuild Specification

## 1. UI Architecture

### Core Components

```typescript
// Window Management System
interface KDXWindow {
  id: string;
  type: WindowType;
  title: string;
  position: Position;
  size: Size;
  content: ReactNode;
  isResizable: boolean;
  isDraggable: boolean;
  theme: WindowTheme;
}

// Custom Window Manager
class WindowManager {
  windows: Map<string, KDXWindow>;
  
  createWindow(config: WindowConfig): string;
  focusWindow(id: string): void;
  arrangeWindows(layout: WindowLayout): void;
  minimizeWindow(id: string): void;
  maximizeWindow(id: string): void;
}

// Theme System
interface WindowTheme {
  primary: string;        // Red theme from screenshots
  secondary: string;      // Dark gray/black
  text: string;          // Lime green for chat
  accent: string;        // Highlight colors
  borders: BorderTheme;
}
```

### Database Schema
```sql
-- Users and Authentication
CREATE TABLE users (
  id UUID PRIMARY KEY,
  username VARCHAR(255) UNIQUE,
  password_hash VARCHAR(255),
  class_id UUID REFERENCES user_classes(id),
  permissions JSONB
);

-- File System
CREATE TABLE files (
  id UUID PRIMARY KEY,
  path VARCHAR(1024),
  parent_id UUID REFERENCES files(id),
  type VARCHAR(50),
  metadata JSONB,
  access_control JSONB
);

-- Chat System
CREATE TABLE chat_rooms (
  id UUID PRIMARY KEY,
  name VARCHAR(255),
  topic TEXT,
  settings JSONB,
  access_control JSONB
);
```

## 5. Build & Deployment

### Client Build
```javascript
// webpack.config.js
module.exports = {
  entry: './src/index.tsx',
  module: {
    rules: [
      {
        test: /\.tsx?$/,
        use: 'ts-loader',
        exclude: /node_modules/,
      },
      {
        test: /\.scss$/,
        use: ['style-loader', 'css-loader', 'sass-loader'],
      }
    ],
  },
  resolve: {
    extensions: ['.tsx', '.ts', '.js'],
  },
};
```

### Server Build
```dockerfile
FROM node:18-alpine

WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npm run build

EXPOSE 10700
CMD ["npm", "start"]
```
